  <script>
    const weddingDate = new Date("2026-06-20T09:30:00");

    function updateCountdown() {
      const now = new Date();
      const diff = weddingDate - now;

      if (diff <= 0) {
        document.getElementById("countdown").innerHTML = "Le grand jour est arrivé ! 🎉";
        return;
      }

      const seconds = Math.floor(diff / 1000);
      document.getElementById("days").textContent = Math.floor(seconds / 86400);
      document.getElementById("hours").textContent = Math.floor((seconds % 86400) / 3600);
      document.getElementById("minutes").textContent = Math.floor((seconds % 3600) / 60);
      document.getElementById("seconds").textContent = seconds % 60;
    }

    updateCountdown();
    setInterval(updateCountdown, 1000);
  </script>

    const scriptURL = "https://script.google.com/macros/s/AKfycbwvUUI1pPvftsCxXq4NIG2pQyQz0ipWLB4ueae2K3dL5xdeKhh9QK4YQa1NwwvRS4Qi/exec";

      const form = new FormData(document.getElementById("rsvpForm"));

      fetch(scriptURL, { method: "POST", body: form })
        .then(() => {

          document.getElementById("rsvpForm").reset();
        });

  <!-- MUSIQUE -->
  <script>
    const bgMusic = document.getElementById("bgMusic");
    const musicToggle = document.getElementById("musicToggle");
    const musicLabel = document.getElementById("musicLabel");
    let musicOn = true;

    window.addEventListener("load", () => {
      bgMusic.volume = 0.4;
      bgMusic.play().catch(() => {
        musicLabel.textContent = "Musique : OFF";
        musicOn = false;
      });
    });

    musicToggle.addEventListener("click", () => {
      if (musicOn) {
        bgMusic.pause();
        musicLabel.textContent = "Musique : OFF";
      } else {
        bgMusic.play();
        musicLabel.textContent = "Musique : ON";
      }
      musicOn = !musicOn;
    });
  </script>
